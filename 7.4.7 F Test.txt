from scipy import stats

# Sample variances for plant growth with two different fertilizers
variance1 = 12
variance2 = 20

# Calculate the F-statistic
f_score = variance2 / variance1

# Degrees of freedom for each sample (assuming equal sample sizes, e.g., 30 each)
dfn = 30 - 1  # degrees of freedom for the numerator
dfd = 30 - 1  # degrees of freedom for the denominator

# Get the p-value
p_value = stats.f.sf(f_score, dfn, dfd)

print("F-score:", f_score)
print("P-value:", p_value)
